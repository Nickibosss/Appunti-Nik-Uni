% --- STILE BLOCCHI DI CODICE BASE ---
% Diciamo a LaTeX di mettere un box scuro attorno a tutti i blocchi "verbatim"
\BeforeBeginEnvironment{verbatim}{
  \vspace{0.2cm}
  \begin{tcolorbox}[
    arc=2mm, 
    boxrule=0.2mm, 
    colback=codebg, 
    colupper=codetext,
    left=2mm, right=2mm, top=2mm, bottom=2mm
  ]
}
\AfterEndEnvironment{verbatim}{
  \end{tcolorbox}
  \vspace{0.2cm}
}

    colframe=calloutborder,
    left=3mm, right=3mm, top=2mm, bottom=2mm
  ]
}{
  \end{tcolorbox}
  \vspace{0.3cm}
}
